function acceptInvite(d){ console.log('acceptInvite called with token:', d && d.token); need( d, ['token','password'] ); const token = String(d.token || '').trim(); const invites = rows('User_Invites'); console.log('Total invites in sheet:', invites.length); const inv = invites.find( x => String(x.token || '').trim() === token && String(x.status || '').trim().toLowerCase() === 'pending' ); if(!inv){ console.error('Invite not found or not pending for token:', token); throw err( 'INVALID_INVITE', 'This invitation is invalid or has already been accepted.' ); } if(Number(inv.expiresAt) < Date.now()){ console.error('Invite expired. Expires at:', inv.expiresAt, 'Current time:', Date.now()); throw err( 'INVALID_INVITE', 'This invitation has expired.' ); } const name = String( d.name||inv.name||'' ).trim(); const email = String( d.email||inv.email||'' ) .trim() .toLowerCase(); if(!name||!email){ throw err( 'VALIDATION_ERROR', 'Name and email are required.' ); } if( !/^[^\s@]+@[^\s@]+\.[^\s@]+$/ .test(email) ){ throw err( 'VALIDATION_ERROR', 'Please enter a valid email address.' ); } if( String(d.password).length<8 ){ throw err( 'VALIDATION_ERROR', 'Password must be at least 8 characters.' ); } if( rows('Users') .some( u => String(u.email||'') .toLowerCase()=== email && String(u.status||'').toLowerCase()==='active' ) ){ throw err( 'EXISTS', 'An active account already exists with that email.' ); } const salt = id(); const userId = id(); add( 'Users', { userId, companyId:inv.companyId, name, email, role:inv.role, status:'active', salt, hash:hash( salt, d.password ), createdAt:now() } ); updateRows( 'User_Invites', x => String(x.inviteId)===String(inv.inviteId), { status:'accepted' } ); if(String(inv.role).toLowerCase()==='admin'){ updateRows( 'Companies', x => String(x.companyId)===String(inv.companyId), { primaryAdminId:userId, primaryAdminEmail:email, status:'active', updatedAt:now(), lastActivityAt:now() } ); } console.log('Account successfully activated for:', email); return { message: 'Account activated. You can log in now.' }; }
